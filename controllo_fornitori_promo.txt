     1	import { useState } from 'react';
     2	import { SafeAreaView } from 'react-native-safe-area-context';
     3	import {
     4	  ActivityIndicator,
     5	  Alert,
     6	  ScrollView,
     7	  StyleSheet,
     8	  Text,
     9	  TextInput,
    10	  TouchableOpacity,
    11	  View,
    12	} from 'react-native';
    13	import * as DocumentPicker from 'expo-document-picker';
    14	import { Feather } from '@expo/vector-icons';
    15	
    16	import { api } from '@/src/api';
    17	import { COLORS, FONTS } from '@/src/theme';
    18	
    19	type PromoFile = {
    20	  uri: string;
    21	  name?: string;
    22	  mimeType?: string;
    23	};
    24	
    25	type PreviewResult = {
    26	  prodotti_letti: number;
    27	  prodotti_trovati: number;
    28	  prodotti_aggiornabili: number;
    29	  prodotti_non_trovati: number;
    30	  prezzo_mancante_o_non_valido: number;
    31	  righe?: Array<{
    32	    riga: number;
    33	    codice_prodotto: string;
    34	    barcode: string;
    35	    descrizione_file: string;
    36	    prezzo_promo: number | null;
    37	    stato: string;
    38	    match_usato: string;
    39	    product_id: string | null;
    40	    descrizione_db: string;
    41	    prezzo_vendita_attuale: number | null;
    42	  }>;
    43	};
    44	
    45	function fmtEUR(value: number | null | undefined) {
    46	  if (value === null || value === undefined || Number.isNaN(Number(value))) return '-';
    47	  return `${Number(value).toFixed(2).replace('.', ',')} €`;
    48	}
    49	
    50	export default function FornitoriScreen() {
    51	  const [file, setFile] = useState<PromoFile | null>(null);
    52	  const [promoNome, setPromoNome] = useState('Promo Stanley 2026');
    53	  const [promoInizio, setPromoInizio] = useState('');
    54	  const [promoFine, setPromoFine] = useState('');
    55	  const [preview, setPreview] = useState<PreviewResult | null>(null);
    56	  const [loading, setLoading] = useState(false);
    57	
    58	  async function scegliFile() {
    59	    try {
    60	      const result = await DocumentPicker.getDocumentAsync({
    61	        type: [
    62	          'text/csv',
    63	          'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    64	          'application/vnd.ms-excel',
    65	          '*/*',
    66	        ],
    67	        copyToCacheDirectory: true,
    68	      });
    69	
    70	      if (result.canceled) return;
    71	
    72	      const asset = result.assets?.[0];
    73	      if (!asset) return;
    74	
    75	      setFile({
    76	        uri: asset.uri,
    77	        name: asset.name,
    78	        mimeType: asset.mimeType,
    79	      });
    80	      setPreview(null);
    81	    } catch (err: any) {
    82	      Alert.alert('Errore file', err?.message || 'Impossibile selezionare il file.');
    83	    }
    84	  }
    85	
    86	  async function anteprimaImport() {
    87	    if (!file) {
    88	      Alert.alert('File mancante', 'Seleziona prima un file CSV o Excel.');
    89	      return;
    90	    }
    91	
    92	    try {
    93	      setLoading(true);
    94	      const res = await api.previewPromoImport(file);
    95	      setPreview(res);
    96	    } catch (err: any) {
    97	      Alert.alert('Errore anteprima', err?.message || 'Import non riuscito.');
    98	    } finally {
    99	      setLoading(false);
   100	    }
   101	  }
   102	
   103	  async function confermaImport() {
   104	    if (!file) {
   105	      Alert.alert('File mancante', 'Seleziona prima un file CSV o Excel.');
   106	      return;
   107	    }
   108	
   109	    if (!promoNome.trim()) {
   110	      Alert.alert('Nome promozione mancante', 'Inserisci il nome della promozione.');
   111	      return;
   112	    }
   113	
   114	    if (!preview) {
   115	      Alert.alert('Anteprima mancante', 'Prima fai l’anteprima, poi conferma.');
   116	      return;
   117	    }
   118	
   119	    Alert.alert(
   120	      'Confermare import?',
   121	      `Aggiornerai ${preview.prodotti_aggiornabili} prodotti con il prezzo promozionale.`,
   122	      [
   123	        { text: 'Annulla', style: 'cancel' },
   124	        {
   125	          text: 'Conferma',
   126	          style: 'destructive',
   127	          onPress: async () => {
   128	            try {
   129	              setLoading(true);
   130	              const res = await api.confirmPromoImport(
   131	                file,
   132	                promoNome.trim(),
   133	                promoInizio.trim(),
   134	                promoFine.trim()
   135	              );
   136	              Alert.alert('Import completato', `Prodotti aggiornati: ${res.aggiornati}`);
   137	              setPreview(res.anteprima);
   138	            } catch (err: any) {
   139	              Alert.alert('Errore conferma', err?.message || 'Aggiornamento non riuscito.');
   140	            } finally {
   141	              setLoading(false);
   142	            }
   143	          },
   144	        },
   145	      ]
   146	    );
   147	  }
   148	
   149	  async function disattivaPromo() {
   150	    Alert.alert(
   151	      'Disattivare promozioni?',
   152	      'Verranno disattivate tutte le promozioni attive. I prezzi standard non verranno modificati.',
   153	      [
   154	        { text: 'Annulla', style: 'cancel' },
   155	        {
   156	          text: 'Disattiva',
   157	          style: 'destructive',
   158	          onPress: async () => {
   159	            try {
   160	              setLoading(true);
   161	              const res = await api.deactivatePromoPrices();
   162	              Alert.alert('Promozioni disattivate', `Prodotti aggiornati: ${res.disattivati}`);
   163	            } catch (err: any) {
   164	              Alert.alert('Errore', err?.message || 'Disattivazione non riuscita.');
   165	            } finally {
   166	              setLoading(false);
   167	            }
   168	          },
   169	        },
   170	      ]
   171	    );
   172	  }
   173	
   174	  const righeProblematiche = preview?.righe?.filter(
   175	    r => r.stato !== 'aggiornabile'
   176	  ).slice(0, 20) || [];
   177	
   178	  return (
   179	    <SafeAreaView style={styles.safe} edges={['top']}>
   180	      <ScrollView contentContainerStyle={styles.container}>
   181	        <View style={styles.header}>
   182	          <View style={styles.iconBox}>
   183	            <Feather name="truck" size={38} color={COLORS.brand} />
   184	          </View>
   185	          <Text style={styles.title}>FORNITORI</Text>
   186	          <Text style={styles.subtitle}>
   187	            Gestione cataloghi, grossisti e promozioni.
   188	          </Text>
   189	        </View>
   190	
   191	        <View style={styles.card}>
   192	          <Text style={styles.cardTitle}>Importa promozione</Text>
   193	          <Text style={styles.cardText}>
   194	            Carica un file CSV o Excel con codice prodotto o barcode e prezzo promo.
   195	            Il prezzo standard resta intatto.
   196	          </Text>
   197	
   198	          <Text style={styles.label}>Nome promozione</Text>
   199	          <TextInput
   200	            style={styles.input}
   201	            value={promoNome}
   202	            onChangeText={setPromoNome}
   203	            placeholder="Es. Promo Stanley 2026"
   204	            placeholderTextColor={COLORS.onSurfaceTertiary}
   205	          />
   206	
   207	          <Text style={styles.label}>Data inizio</Text>
   208	          <TextInput
   209	            style={styles.input}
   210	            value={promoInizio}
   211	            onChangeText={setPromoInizio}
   212	            placeholder="Es. 2026-01-01"
   213	            placeholderTextColor={COLORS.onSurfaceTertiary}
   214	          />
   215	
   216	          <Text style={styles.label}>Data fine</Text>
   217	          <TextInput
   218	            style={styles.input}
   219	            value={promoFine}
   220	            onChangeText={setPromoFine}
   221	            placeholder="Es. 2026-12-31"
   222	            placeholderTextColor={COLORS.onSurfaceTertiary}
   223	          />
   224	
   225	          <TouchableOpacity style={styles.secondaryButton} onPress={scegliFile}>
   226	            <Feather name="file-plus" size={18} color={COLORS.onSurface} />
   227	            <Text style={styles.secondaryButtonText}>
   228	              {file?.name ? file.name : 'Seleziona file CSV / Excel'}
   229	            </Text>
   230	          </TouchableOpacity>
   231	
   232	          <View style={styles.row}>
   233	            <TouchableOpacity
   234	              style={[styles.button, loading && styles.disabled]}
   235	              onPress={anteprimaImport}
   236	              disabled={loading}
   237	            >
   238	              <Text style={styles.buttonText}>ANTEPRIMA</Text>
   239	            </TouchableOpacity>
   240	
   241	            <TouchableOpacity
   242	              style={[styles.button, styles.confirmButton, loading && styles.disabled]}
   243	              onPress={confermaImport}
   244	              disabled={loading}
   245	            >
   246	              <Text style={styles.buttonText}>CONFERMA</Text>
   247	            </TouchableOpacity>
   248	          </View>
   249	
   250	          {loading && <ActivityIndicator style={{ marginTop: 16 }} />}
   251	        </View>
   252	
   253	        {preview && (
   254	          <View style={styles.card}>
   255	            <Text style={styles.cardTitle}>Risultato anteprima</Text>
   256	
   257	            <View style={styles.statRow}>
   258	              <Text style={styles.statLabel}>Prodotti letti</Text>
   259	              <Text style={styles.statValue}>{preview.prodotti_letti}</Text>
   260	            </View>
   261	            <View style={styles.statRow}>
   262	              <Text style={styles.statLabel}>Prodotti trovati</Text>
   263	              <Text style={styles.statValue}>{preview.prodotti_trovati}</Text>
   264	            </View>
   265	            <View style={styles.statRow}>
   266	              <Text style={styles.statLabel}>Aggiornabili</Text>
   267	              <Text style={styles.statValue}>{preview.prodotti_aggiornabili}</Text>
   268	            </View>
   269	            <View style={styles.statRow}>
   270	              <Text style={styles.statLabel}>Non trovati</Text>
   271	              <Text style={styles.statValue}>{preview.prodotti_non_trovati}</Text>
   272	            </View>
   273	            <View style={styles.statRow}>
   274	              <Text style={styles.statLabel}>Prezzo mancante/non valido</Text>
   275	              <Text style={styles.statValue}>{preview.prezzo_mancante_o_non_valido}</Text>
   276	            </View>
   277	          </View>
   278	        )}
   279	
   280	        {preview?.righe?.length ? (
   281	          <View style={styles.card}>
   282	            <Text style={styles.cardTitle}>Prime righe aggiornabili</Text>
   283	            {preview.righe.filter(r => r.stato === 'aggiornabile').slice(0, 10).map((r, index) => (
   284	              <View key={`${r.riga}-${index}`} style={styles.productRow}>
   285	                <Text style={styles.productTitle}>{r.descrizione_db || r.descrizione_file}</Text>
   286	                <Text style={styles.productText}>Codice: {r.codice_prodotto || '-'}</Text>
   287	                <Text style={styles.productText}>Barcode: {r.barcode || '-'}</Text>
   288	                <Text style={styles.productText}>Match: {r.match_usato || '-'}</Text>
   289	                <Text style={styles.promoPrice}>Promo: {fmtEUR(r.prezzo_promo)}</Text>
   290	              </View>
   291	            ))}
   292	          </View>
   293	        ) : null}
   294	
   295	        {righeProblematiche.length > 0 && (
   296	          <View style={styles.card}>
   297	            <Text style={styles.cardTitle}>Righe da controllare</Text>
   298	            {righeProblematiche.map((r, index) => (
   299	              <View key={`${r.riga}-${index}`} style={styles.problemRow}>
   300	                <Text style={styles.productTitle}>Riga {r.riga}: {r.stato}</Text>
   301	                <Text style={styles.productText}>Codice: {r.codice_prodotto || '-'}</Text>
   302	                <Text style={styles.productText}>Barcode: {r.barcode || '-'}</Text>
   303	                <Text style={styles.productText}>Descrizione: {r.descrizione_file || '-'}</Text>
   304	              </View>
   305	            ))}
   306	          </View>
   307	        )}
   308	
   309	        <View style={styles.card}>
   310	          <Text style={styles.cardTitle}>Disattiva promozioni</Text>
   311	          <Text style={styles.cardText}>
   312	            Disattiva tutte le promo attive senza cancellare i prezzi standard.
   313	          </Text>
   314	          <TouchableOpacity style={styles.dangerButton} onPress={disattivaPromo} disabled={loading}>
   315	            <Text style={styles.buttonText}>DISATTIVA PROMO</Text>
   316	          </TouchableOpacity>
   317	        </View>
   318	      </ScrollView>
   319	    </SafeAreaView>
   320	  );
   321	}
